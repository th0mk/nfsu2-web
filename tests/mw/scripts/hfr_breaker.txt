# covered per second of game time over the speed) compare across rates.  A world steps at 60 unless

#   NFSU2_SAVE_SEED=DIR/new node web/tools/run-suite.mjs --game mw --script hfr_breaker.txt \
#     --marker 'hfr_breaker: refilled' --env NFSU2_FPS=120 --env NFSMW_WORLD_HFR=1 --env NFSMW_HFR_TRACE=1 \
#     --timeout-scale 8
