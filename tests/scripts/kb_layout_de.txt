# NFSU2_KEYLAYOUT=com.apple.keylayout.German.  WM_CHAR carries the layout's

# character with: "Z" is the key in the US Y position (DIK_Y), "y" the one
# in the US Z position (DIK_Z); "é" is the dead key ´ followed by e
# (WM_DEADCHAR 0xB4, then WM_CHAR 0xE9).
