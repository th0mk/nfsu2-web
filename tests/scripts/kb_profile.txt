# Real keyboard path (synthesised NSEvents through metal.m's key handler,
# runtime/keyboard_mac.m): create a profile by typing its name with mixed
# case, digits and spaces (the game's name field keeps letters and digits
# only, 7 of them: 0x4ff310, [0x83a9b4] = field type, [0x83a9bc] = length),
# delete with an autorepeating Backspace, type with Caps Lock, save it (the
# save dialog starts on "No": Left selects "Yes"; the profile lands in the
# run's save/ directory), open the 'Q'
# quit prompt from the main menu and leave it with Escape, then quit with
# Cmd-Q, which the game must not see as a 'q' (it gets WM_CLOSE from the
# menu instead).
0     nstapuntil RETURN MC_Background.fng 120 4000  # title -> "create a new profile?"
+120  nstap LEFT                                    # "Yes"
+30   nstap RETURN
+240  dump name_default
+0    nstap BACK                                    # Delete = Backspace: clear "NAME"
+8    nstap BACK
+8    nstap BACK
+8    nstap BACK
+20   nstype Tobi M4 Pro
+80   dump name_typed
+0    peek 0x83a9b4 3
+0    nsdown BACK                                   # held: 1 + 3 autorepeats
+30   nsrepeat BACK 3
+10   nsup BACK
+40   dump name_after_repeat
+0    nstap CAPSLOCK                                # Caps Lock on: capitals without Shift
+20   nstype ias
+20   nstap CAPSLOCK                                # off again
+20   nstype 1                                      # -> "TobIAS1"
+40   dump name_retyped
+0    nstap RETURN                                  # -> "save profile named ...?" ("No" selected)
+240  dump name_confirm
+0    nstap LEFT                                    # "Yes"
+30   nstap RETURN
+240  dump save_complete
+0    nstapuntil RETURN GarageMain.fng 150 3000     # "Save complete." Continue -> main menu
+300  dump mainmenu
+0    nstap Q                                       # the game's quit prompt (WM_CHAR 'q')
+150  dump quit_prompt
+0    nstap ESCAPE                                  # cancel it
+150  dump after_escape
+0    log Cmd-Q now
+0    nstap cmd+Q
+300  dump after_cmdq
+600  quit
